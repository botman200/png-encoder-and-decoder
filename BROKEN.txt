    <h1>Universal Packer</h1>

    <hr>

    <!-- PACK SECTION -->
    <h2>1. Pack Files to Text</h2>
    <p><strong>WARNING:</strong> Browser engine limitations cap file sizes at roughly <strong>2 GB</strong>. Do not use this section for full-length movies or massive folders over this limit.</p>
    
    <form>
        <input type="file" id="fileInput" multiple><br><br>
        <button type="button" id="packBtn" style="display:none;">Pack to Text File</button>
    </form>
    <p id="packStatus"></p>

    <hr>

    <!-- UNPACK SECTION -->
    <h2>2. Unpack Text back to Files</h2>
    <p><strong>WARNING:</strong> Ensure the text file you load here was generated from an archive smaller than 2 GB.</p>
    
    <form>
        <input type="file" id="textInput" accept=".txt"><br><br>
        <button type="button" id="unpackBtn" style="display:none;">Unpack Archive</button>
    </form>
    <p id="unpackStatus"></p>

    <hr>

    <script>
        let filesToPack = null;
        let fileToUnpack = null;

        document.getElementById('fileInput').addEventListener('change', (e) => {
            filesToPack = e.target.files;
            document.getElementById('packBtn').style.display = filesToPack.length ? 'inline' : 'none';
        });

        document.getElementById('textInput').addEventListener('change', (e) => {
            fileToUnpack = e.target.files;
            document.getElementById('unpackBtn').style.display = fileToUnpack ? 'inline' : 'none';
        });

        // ENCODER
        document.getElementById('packBtn').addEventListener('click', async () => {
            const status = document.getElementById('packStatus');
            status.innerText = "Processing stream...";
            
            let fileEntries = [];
            let totalPayloadSize = 0;
            
            // Step 1: Read all files and compile metadata size
            for (let file of filesToPack) {
                if (file.size > 2 * 1024 * 1024 * 1024) {
                    alert(`Error: ${file.name} is too large. Choose files under 2 GB.`);
                    status.innerText = "";
                    return;
                }
                const buf = await file.arrayBuffer();
                const fileBytes = new Uint8Array(buf);
                const nameBytes = new TextEncoder().encode(file.name);
                
                fileEntries.push({
                    nameBytes: nameBytes,
                    fileBytes: fileBytes,
                    size: fileBytes.length
                });
                
                // Each metadata entry takes: 4 bytes (name len) + name string bytes + 4 bytes (file len) + file content bytes
                totalPayloadSize += 4 + nameBytes.length + 4 + fileBytes.length;
            }
            
            // Add 4 bytes at the very beginning to store the total number of files packed
            let masterArray = new Uint8Array(4 + totalPayloadSize);
            let view = new DataView(masterArray.buffer);
            
            // Set number of files
            view.setUint32(0, fileEntries.length, false);
            let offset = 4;
            
            // Step 2: Write pure binary header segments alongside data blocks
            for (let entry of fileEntries) {
                // Write Name Length (4 bytes)
                view.setUint32(offset, entry.nameBytes.length, false);
                offset += 4;
                
                // Write Name String
                masterArray.set(entry.nameBytes, offset);
                offset += entry.nameBytes.length;
                
                // Write File Content Length (4 bytes)
                view.setUint32(offset, entry.size, false);
                offset += 4;
                
                // Write File Bytes
                masterArray.set(entry.fileBytes, offset);
                offset += entry.size;
            }

            status.innerText = "Compressing blocks...";
            
            const compressionStream = new Blob([masterArray]).stream().pipeThrough(new CompressionStream('gzip'));
            const compBuffer = await new Response(compressionStream).arrayBuffer();
            
            let binary = '';
            const bytes = new Uint8Array(compBuffer);
            for (let i = 0; i < bytes.length; i += 0xffff) {
                binary += String.fromCharCode.apply(null, bytes.subarray(i, i + 0xffff));
            }
            const b64Text = btoa(binary);

            const anchor = document.createElement('a');
            anchor.href = URL.createObjectURL(new Blob([b64Text], {type: "text/plain"}));
            anchor.download = "packed_archive.txt";
            anchor.click();
            status.innerText = "Done. Exported to downloads.";
        });

        // DECODER
        document.getElementById('unpackBtn').addEventListener('click', () => {
            const status = document.getElementById('unpackStatus');
            status.innerText = "Reading file content...";
            
            const reader = new FileReader();
            reader.onload = async (e) => {
                try {
                    const b64Text = e.target.result.trim();
                    const binary = atob(b64Text);
                    const compBytes = new Uint8Array(binary.length);
                    for (let i = 0; i < binary.length; i++) compBytes[i] = binary.charCodeAt(i);
                    
                    status.innerText = "Decompressing data...";
                    const decompStream = new Blob([compBytes]).stream().pipeThrough(new DecompressionStream('gzip'));
                    const decompBuffer = await new Response(decompStream).arrayBuffer();
                    const masterArray = new Uint8Array(decompBuffer);
                    let view = new DataView(masterArray.buffer);
                    
                    // Parse total file count from first 4 bytes
                    const fileCount = view.getUint32(0, false);
                    let offset = 4;
                    
                    status.innerText = "Extracting binary sequences...";
                    
                    for (let i = 0; i < fileCount; i++) {
                        // Read name length
                        const nameLen = view.getUint32(offset, false);
                        offset += 4;
                        
                        // Read name string
                        const nameBytes = masterArray.subarray(offset, offset + nameLen);
                        const name = new TextDecoder().decode(nameBytes);
                        offset += nameLen;
                        
                        // Read file data length
                        const fileLen = view.getUint32(offset, false);
                        offset += 4;
                        
                        // Extract precise file data subset slice
                        const fileBytes = masterArray.subarray(offset, offset + fileLen);
                        offset += fileLen;
                        
                        // Trigger individual item download anchor
                        const anchor = document.createElement('a');
                        anchor.href = URL.createObjectURL(new Blob([fileBytes], {type: "application/octet-stream"}));
                        anchor.download = name;
                        anchor.click();
                    }
                    
                    status.innerText = "Success. All contents extracted.";
                } catch(err) {
                    alert("Error processing the data stream.");
                    status.innerText = "";
                }
            };
            reader.readAsText(fileToUnpack);
        });
    </script>
